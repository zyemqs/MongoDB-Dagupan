Criterion 5 — Runtime Error Fix

Runtime Error
Faulty Code
function getLength(str) {
  return str.length;  // ❌ TypeError if str is null/undefined
}
Fixed Code
function getLength(str) {
  if (str === null || str === undefined) {  // ✅ null guard
    console.log("getLength: received null or undefined — returning 0");
    return 0;
  }
  return str.length;
}

Length of 'hello': 5

Criteria 2 & 3 — testFunction: Division-by-Zero Guard

Runtime-Safe Fix
Faulty Code
function testFunction(a, b) {
  return a / b;  // ❌ returns Infinity or NaN when b === 0
}
Fixed Code
function testFunction(a, b) {
  console.log("testFunction called with a =", a, "and b =", b);
  if (b !== 0) {                          // ✅ division-by-zero guard
    const result = a / b;
    console.log("Result:", result);
    return result;
  } else {
    console.log("Error: Division by zero is not allowed.");
    return null;
  }
}
// Expected console output:
testFunction called with a = 10 and b = 2
Result: 5
testFunction called with a = 9 and b = 3
Result: 3
testFunction called with a = 7 and b = 0
Error: Division by zero is not allowed.
